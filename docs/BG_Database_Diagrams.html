BG Automation — Database Diagrams (PostgreSQL)

Conceptual ER (key attributes, crow's foot) followed by the full logical schema. Table design matches db/migrations/V1__schema.sql plus db/migrations/V3__schema.sql; seed values match db/migrations/V2__seed.sql and docs/master-data.md. One BG row per guarantee, dedupe on bg_no; extensions chain via supersedes_bg_id under the same loa_id (1 LOA : M BGs). Dates are DATE; instants are TIMESTAMPTZ in UTC.

    COMPANY {
        SMALLINT id PK
        TEXT name UK "e.g. NSPPL"

        BOOLEAN is_active
    }
    PROJECT {
        SMALLINT id PK
        SMALLINT company_id FK
        TEXT code UK "e.g. SB, MH"
        TEXT name
        TEXT state
        INTEGER round
        TEXT project_manager
        TEXT source
        BOOLEAN is_active
    }
    TOLL_PLAZA {
        INTEGER id PK
        SMALLINT project_id FK
        TEXT name "UNIQUE per project"
        TEXT chainage
        TEXT toll_manager
        TEXT maintenance_manager
        TEXT corridor_manager
        TEXT source
        BOOLEAN is_active
    }
    BANK {
        SMALLINT id PK
        TEXT name UK "created by the first BG"
        TEXT source
    }
    CONTRACTOR {
        INTEGER id PK
        TEXT name UK "created by the first BG"
        TEXT source
    }
    BG_TYPE {
        TEXT code PK "PBG, ABG"
        INTEGER sort_order
    }
    NATURE_OF_SERVICE {
        SMALLINT id PK
        TEXT name UK "27 master values"
    }
    BG_STATUS {
        TEXT code PK "Draft..Closed"
        INTEGER sort_order
    }
    ROLES {
        SMALLINT id PK
        TEXT name UK "admin, uploader, viewer"
    }
    USERS {
        UUID id PK
        CITEXT username UK "case-insensitive"
        CITEXT email UK "case-insensitive, nullable"
        TEXT full_name
        TEXT password_hash "scrypt"
        BOOLEAN is_active
        TIMESTAMPTZ last_login
        TIMESTAMPTZ created_at
        TIMESTAMPTZ updated_at
        INTEGER failed_attempts "lockout after 5"
        TIMESTAMPTZ locked_until
    }
    USER_ROLES {
        UUID user_id PK, FK
        SMALLINT role_id PK, FK
    }
    SESSIONS {

        TEXT user_agent

    LOA_RECORDS {

        UUID created_by FK

        TIMESTAMPTZ updated_at

    BG_RECORDS {
        UUID id PK

        SMALLINT project_id FK

        SMALLINT nature_id FK
        INTEGER contractor_id FK

        TEXT po_number
        NUMERIC bg_amount "CHECK > 0"

        DATE to_date "T-7 anchor"

        DATE extended_date "NULL until extended"
        TEXT status FK "default Draft"

        TEXT source_pdf
        UUID created_by FK
        TIMESTAMPTZ created_at
        TIMESTAMPTZ updated_at
        TIMESTAMPTZ reminder_sent_at

    }
    OCR_RESULTS {

    REMINDERS {
